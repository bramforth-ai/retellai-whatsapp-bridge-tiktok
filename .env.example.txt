# ============================================
# SERVER CONFIGURATION
# ============================================

# Port the server runs on (default: 5000)
PORT=5000

# Environment mode: 'development' or 'production'
# - development: Chat widget enabled, no API key required, Twilio validation skipped
# - production: Chat widget hidden, API key required, Twilio validation enforced
NODE_ENV=development


# ============================================
# RETELLAI CONFIGURATION
# ============================================

# Your RetellAI API key (starts with 'key_')
# Get from: https://dashboard.retellai.com/ → Settings → API Keys
RETELL_API_KEY=key_your_retell_api_key_here

# Your RetellAI Chat Agent ID (starts with 'agent_')
# Get from: https://dashboard.retellai.com/ → Agents → Your Agent
# Note: Must be a CHAT agent, not a voice agent
RETELL_AGENT_ID=agent_your_agent_id_here


# ============================================
# MONGODB CONFIGURATION
# ============================================

# MongoDB Atlas connection string
# Format: mongodb+srv://username:password@cluster.mongodb.net/
# Get from: MongoDB Atlas → Database → Connect → Connect your application
# Important: Replace <password> with your actual password
MONGODB_URI=mongodb+srv://username:password@cluster.mongodb.net/

# Database name (can be any name you choose)
# Example: retell_whatsapp, whatsapp_bridge, etc.
MONGODB_DATABASE_NAME=retell_whatsapp


# ============================================
# TWILIO CONFIGURATION
# ============================================

# Your Twilio Account SID (starts with 'AC')
# Get from: https://console.twilio.com/ → Dashboard
TWILIO_ACCOUNT_SID=ACxxxxxxxxxxxxxxxxxxxxxxxxxxxxx

# Your Twilio Auth Token
# Get from: https://console.twilio.com/ → Dashboard
TWILIO_AUTH_TOKEN=your_twilio_auth_token_here

# Production example: whatsapp:+447723451913

TWILIO_WHATSAPP_NUMBER=whatsapp:+14155238886
